#define UP_BUTTON     42
#define DOWN_BUTTON   41
#define RIGHT_BUTTON  2
#define LEFT_BUTTON   1
#define STOP_BUTTON   40

// digit sent after TEAM_TAG for each button (RX accepts '1'-'7')
#define UP_COMMAND    '1'
#define DOWN_COMMAND  '2'
#define RIGHT_COMMAND '3'
#define LEFT_COMMAND  '4'
#define STOP_COMMAND  '5'

volatile char pendingCommand = 0;  // 0 = nothing to send, set by the ISRs
volatile unsigned long lastInterruptTime = 0;

const char* SERVER_IP = "10.5.9.24";  // IP of server ESP32
const uint16_t SERVER_PORT = 80;
const char* SERVER_PATH = "/ws";

const char* WIFI_SSID = SECRET_SSID;
const char* WIFI_PASSWORD = SECRET_PASS;

const char* CLIENT_ID = "BLUESCLUES12"; //esp without breadboard

unsigned long lastSendTime = 0;

enum State {
    STATE_1,
    STATE_2,
    STATE_3,
    STATE_4,
    STATE_5,
    STATE_6,
    STATE_7,
    NUM_STATES
};

Websocket webSocket(
    WIFI_SSID,
    WIFI_PASSWORD,
    SERVER_IP,
    SERVER_PORT,
    SERVER_PATH,
    CLIENT_ID
);

void IRAM_ATTR queueCommand(char command) { //shared by all button ISRs
  unsigned long now = millis();
  if (now - lastInterruptTime > 200) {  // debounce
    pendingCommand = command;
    lastInterruptTime = now;
  }
}

//One Interrupt Service Routine per pushbutton
void IRAM_ATTR handleUpISR()    { 
    queueCommand(UP_COMMAND); 
}
void IRAM_ATTR handleDownISR()  { 
    queueCommand(DOWN_COMMAND); 
}
void IRAM_ATTR handleRightISR() { 
    queueCommand(RIGHT_COMMAND); 
}
void IRAM_ATTR handleLeftISR()  { 
    queueCommand(LEFT_COMMAND); 
}
void IRAM_ATTR handleStopISR()  { 
    queueCommand(STOP_COMMAND); 
}

State currentState = STATE_1;

void setup() {
    Serial.begin(115200);
    webSocket.begin();

    delay(1000);

    //MOTOR PINS configure direction pins to be output
    // pinMode(MOTOR_1_DIR1, OUTPUT);
    // pinMode(MOTOR_1_DIR2, OUTPUT);
    // pinMode(MOTOR_2_DIR1, OUTPUT);
    // pinMode(MOTOR_2_DIR2, OUTPUT);

    //BUTTON PINs for directions
    pinMode(UP_BUTTON, INPUT_PULLUP);
    pinMode(DOWN_BUTTON, INPUT_PULLUP);
    pinMode(RIGHT_BUTTON, INPUT_PULLUP);
    pinMode(LEFT_BUTTON, INPUT_PULLUP);
    pinMode(STOP_BUTTON, INPUT_PULLUP);

    attachInterrupt(digitalPinToInterrupt(UP_BUTTON), handleUpISR, FALLING); //GPIO pin, ISR, event
    attachInterrupt(digitalPinToInterrupt(DOWN_BUTTON), handleDownISR, FALLING); //GPIO pin, ISR, event
    attachInterrupt(digitalPinToInterrupt(RIGHT_BUTTON), handleRightISR, FALLING); //GPIO pin, ISR, event
    attachInterrupt(digitalPinToInterrupt(LEFT_BUTTON), handleLeftISR, FALLING); //GPIO pin, ISR, event
    attachInterrupt(digitalPinToInterrupt(STOP_BUTTON), handleStopISR, FALLING);
}

void loop() {
    //delay(1000);
    webSocket.loop();

    if (pendingCommand != 0) {
        char command = pendingCommand;
        pendingCommand = 0;

        String myMessage = String(TEAM_TAG) + command;
        Serial.println(myMessage);
        if (!webSocket.sendText(myMessage)) {
            Serial.println("not sent (not authenticated)");
        }
    }
}
